import {chromium} from 'playwright';
import {readFile,writeFile} from 'node:fs/promises';
import {once} from 'node:events';
import {createHash} from 'node:crypto';
import {loadVersion,inside} from '../../scripts/project.mjs';
import {createProjectServer} from '../../scripts/server.mjs';
const v=await loadVersion('anabology','v001.1'),m=v.manifest;
if(m.status!=='review'||!m.variants[0].output)throw Error('Final output must be registered.');
const server=createProjectServer(v.projectDirectory);server.listen(0,'127.0.0.1');await once(server,'listening');let browser;
try{
 browser=await chromium.launch({headless:true,channel:'chrome'});const page=await browser.newPage({viewport:{width:1600,height:1000}}),errors=[];page.on('pageerror',e=>errors.push(e.message));
 await page.goto(`http://127.0.0.1:${server.address().port}/versions/v001.1/watch.html`);
 await page.waitForFunction(()=>document.querySelector('#film').readyState>=1&&!document.querySelector('#full').disabled);
 const player=page.locator('#film'),read=()=>player.evaluate(v=>({time:v.currentTime,readyState:v.readyState,error:v.error?.message??null,paused:v.paused,muted:v.muted,width:v.videoWidth,height:v.videoHeight,duration:v.duration}));
 const responsive=[];for(const size of [{width:1600,height:1000},{width:1024,height:768},{width:390,height:844}]){await page.setViewportSize(size);const s=await page.evaluate(()=>({viewport:[innerWidth,innerHeight],horizontal_overflow:document.documentElement.scrollWidth>innerWidth,source:document.querySelector('#film').currentSrc,full_enabled:!document.querySelector('#full').disabled,preview_enabled:!document.querySelector('#preview').disabled,full_pressed:document.querySelector('#full').getAttribute('aria-pressed')}));if(s.horizontal_overflow||!s.full_enabled||s.full_pressed!=='true')throw Error('4K watch controls or responsive layout failed.');responsive.push(s);}
 await player.evaluate(async v=>{v.muted=true;await v.play();});await page.waitForTimeout(1200);const initial=await read();if(initial.time<=0||initial.readyState<3||initial.error||initial.width!==3840||initial.height!==2160)throw Error('Final 4K playback failed.');
 await player.evaluate((v,t)=>new Promise(resolve=>{v.addEventListener('seeked',resolve,{once:true});v.currentTime=t}),65.2);
 const dense=[];for(let i=0;i<10;i++){await page.waitForTimeout(600);const state=await read();if(state.readyState<3||state.error||state.paused)throw Error('Dense-segment playback failed.');dense.push(state);}
 if(dense.at(-1).time<=65.2+3)throw Error('Dense segment clock did not advance.');
 await page.locator('#chapter').selectOption('216.52');await page.waitForTimeout(350);const chapter=await read();if(chapter.time<216.52||chapter.time>218||chapter.error||chapter.readyState<3)throw Error('Chapter selection failed.');
 await player.evaluate(async v=>{v.muted=false;await v.play();});const before=await read();await page.waitForTimeout(1200);const after=await read();if(after.time<=before.time||after.error||after.muted||after.paused)throw Error('Unmuted playback failed.');
 await player.evaluate((v,t)=>new Promise(resolve=>{v.addEventListener('seeked',resolve,{once:true});v.currentTime=t}),303.5);
 await page.waitForFunction(()=>document.querySelector('#film').ended,{timeout:10000});const tail=await player.evaluate(v=>({time:v.currentTime,ended:v.ended,error:v.error?.message??null,duration:v.duration}));if(!tail.ended||tail.error||Math.abs(tail.time-tail.duration)>.1)throw Error('Full tail playback failed.');
 if(errors.length)throw Error(errors.join('\n'));
 const q={project_id:m.project_id,version:m.version,generation_id:m.variants[0].generation_id,run_id:m.variants[0].output.run_id,date:new Date().toISOString(),status:'passed',browser:browser.version(),initial,dense_segment:dense,chapter_select:chapter,unmuted_playback:{before,after,claim:'technical playback only; perceptual sync pending human listening'},tail,responsive,errors,qa_tool_sha256:createHash('sha256').update(await readFile(new URL(import.meta.url))).digest('hex'),full_video_human_approval:null};
 await writeFile(inside(v.directory,'provenance/final-watch-qa.json'),JSON.stringify(q,null,2)+'\n',{flag:'wx'});
 await writeFile(inside(v.directory,'provenance/final-watch-qa-tool.mjs.txt'),await readFile(new URL(import.meta.url)),{flag:'wx'});
 console.log(JSON.stringify({status:'passed',actual_profile:[initial.width,initial.height],dense_elapsed:dense.at(-1).time-65.2,tail_ended:tail.ended,responsive_viewports:responsive.length}));
}finally{if(browser)await browser.close();server.close();}
